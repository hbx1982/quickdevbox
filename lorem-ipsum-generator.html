Implementation & Code Examples

JavaScript Placeholder Generator

const LOREM_WORDS = ["lorem", "ipsum", "dolor", "sit", "amet", "consectetur", "adipiscing", "elit", "sed", "do", "eiusmod", "tempor", "incididunt"];

function generateLoremWords(count = 20) {
  const result = [];
  for (let i = 0; i < count; i++) {
    result.push(LOREM_WORDS[i % LOREM_WORDS.length]);
  }
  return result.join(' ') + '.';
}
console.log(generateLoremWords(10));
